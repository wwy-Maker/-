## 1. 测试策略

### 1.1 测试框架

- **框架**: Unity Test Framework（UTF），基于 NUnit
- **包**: `com.unity.test-framework`（Unity 2022.3 LTS 内置）
- **测试类型**: EditMode Tests（纯逻辑，不依赖场景）+ PlayMode Tests（需要 MonoBehaviour/物理/场景）
- **运行方式**: Unity Editor 内 Test Runner 窗口 / 命令行 batchmode

### 1.2 测试层级

| 层级 | 类型 | 覆盖范围 | 对应 Story |
|------|------|---------|-----------|
| 1. 配置校验测试 | EditMode | 所有 JSON 配置格式正确、必填字段齐全、数值在合理范围、引用完整性 | S05, S07, S13, S20, S22, S23, S30, S43, S46 |
| 2. 战斗公式测试 | EditMode | 伤害公式 5 层计算、反弹公式、边界值 | S15 |
| 3. 输入分流测试 | EditMode | Shift 行为分流（非御艺=闪避/御艺=冲刺/Lv4分支A=纯冲刺）、体力检查、冷却检查 | S08, S18 |
| 4. 状态机测试 | EditMode | 状态转换规则、非法转换拦截 | S09 |
| 5. Foundation 测试 | EditMode | ObjectPool 复用、EventBus 订阅/发布、ServiceLocator 注册/获取、SaveSystem 读写 | S03, S04 |
| 6. 弹幕模式测试 | PlayMode | 8 种弹幕模式的生成、运动、碰撞 | S16 |
| 7. Boss 阶段测试 | PlayMode | Boss 阶段切换、弹幕模式轮转、特殊机制触发 | S31, S32, S33, S34 |
| 8. 经济系统测试 | EditMode | 学识掉落计算、升级费用、余额不足拒绝 | S24, S36 |
| 9. 灰阶可辨测试 | PlayMode + 人工 | 去色后弹幕可辨性、波纹vs反弹圈区分 | S49 |
| 10. 手感测试 | 人工 Playtest | 非自动化，用 GDD 手感验证标准 | S26, S42, S51 |

### 1.3 测试覆盖率目标

| 优先级 | 覆盖率目标 | 说明 |
|--------|-----------|------|
| P0 Story | **100% 有 EditMode 测试** | 核心路径必须可自动验证 |
| P1 Story | **70% 有 EditMode 或 PlayMode 测试** | Boss/武器行为关键路径覆盖 |
| P2 Story | 不强制 | 器物/难度/死亡面板，冒烟测试覆盖即可 |

### 1.4 验证驱动开发（VDD）流程

```
1. 读 Story 验收标准
2. 先写测试（红灯——测试失败因为功能未实现）
3. 实现功能代码（绿灯——测试通过）
4. 重构（保持绿灯）
5. 测试结果存入 Tests/results/<story-id>/
```

// 测试用 InputReader 模拟，不依赖物理按键
public class MockInputReader : IInputReader
{
    private Queue<InputEvent> _queuedEvents = new Queue<InputEvent>();

    public void QueueEvent(InputEvent inputEvent)
    {
        _queuedEvents.Enqueue(inputEvent);
    }

    public List<InputEvent> Poll()
    {
        var events = new List<InputEvent>();
        while (_queuedEvents.Count > 0)
            events.Add(_queuedEvents.Dequeue());
        return events;
    }
}

// 测试用 PlayerSystem 模拟，可设置任意状态

}

using NUnit.Framework;
using UnityEngine;

/// 配置校验测试 — 对应 Story: S05, S07, S13, S20, S22, S23, S30, S43, S46
/// 验证所有 JSON 配置文件格式正确、必填字段齐全、数值在合理范围、引用完整性

[TestFixture]
public class ConfigValidationTests
{
    private ConfigLoader _loader;
    private ConfigValidator _validator;

    [SetUp]
    public void Setup()
    {
        _loader = new ConfigLoader();
        _loader.SetConfigRoot("Assets/Configs/");
        _validator = new ConfigValidator(_loader);
    }

    // ===== bullets.json 校验 =====

    [Test]
    public void BulletsConfig_AllBulletIds_Unique()
    {
        var config = _loader.GetConfig<BulletConfigCollection>();
        var ids = config.bullets.Select(b => b.bulletId).ToList();
        var duplicates = ids.GroupBy(x => x).Where(g => g.Count() > 1).ToList();
        Assert.IsEmpty(duplicates, $"Duplicate bulletIds found: {string.Join(", ", duplicates.Select(g => g.Key))}");
    }

    [Test]
    public void BulletsConfig_AllSpeeds_Positive()
    {
        var config = _loader.GetConfig<BulletConfigCollection>();
        foreach (var bullet in config.bullets)
        {
            // DashDamageZone 的 speed 可以为 0（静态区域）
            if (bullet.pattern != "DashDamageZone" && bullet.pattern != "BounceRing")
            {
                Assert.Greater(bullet.speed, 0f, 
                    $"Bullet {bullet.bulletId} has non-positive speed: {bullet.speed}");
            }
        }
    }

    [Test]
    public void BulletsConfig_AllDamage_NonNegative()
    {
        var config = _loader.GetConfig<BulletConfigCollection>();
        foreach (var bullet in config.bullets)
        {
            Assert.GreaterOrEqual(bullet.damage, 0f,
                $"Bullet {bullet.bulletId} has negative damage: {bullet.damage}");
        }
    }

    [Test]
    public void BulletsConfig_AllLifetimes_Positive()
    {
        var config = _loader.GetConfig<BulletConfigCollection>();
        foreach (var bullet in config.bullets)
        {
            Assert.Greater(bullet.lifetime, 0f,
                $"Bullet {bullet.bulletId} has non-positive lifetime: {bullet.lifetime}");
        }
    }

    [Test]
    public void BulletsConfig_RippleRing_HasRingParameters()
    {
        var config = _loader.GetConfig<BulletConfigCollection>();
        var ripples = config.bullets.Where(b => b.pattern == "RippleRing").ToList();

            Assert.Greater(ripple.ringExpandSpeed, 0f,

                $"RippleRing {ripple.bulletId} missing ringMaxRadius");

            .Where(b => b.pattern == "RippleRing" || b.pattern == "BounceRing")
            .ToList();
        foreach (var ring in rings)
        {
            Assert.Greater(ring.lineWidth, 0f,
                $"Ring bullet {ring.bulletId} missing lineWidth for grayscale discernibility");
        }
    }

    // ===== weapons.json 校验 =====

    [Test]
    public void WeaponsConfig_AllWeapons_HaveFiveLevels()
    {
        var config = _loader.GetConfig<WeaponConfigCollection>();
        foreach (var weapon in config.weapons)
        {
            Assert.AreEqual(5, weapon.levels.Length,
                $"Weapon {weapon.weaponId} does not have exactly 5 levels");
        }
    }

    [Test]
    public void WeaponsConfig_AllBulletIds_ExistInBulletsConfig()
    {
        var weapons = _loader.GetConfig<WeaponConfigCollection>();
        var bullets = _loader.GetConfig<BulletConfigCollection>();
        var bulletIds = bullets.bullets.Select(b => b.bulletId).ToHashSet();

        foreach (var weapon in weapons.weapons)
        {

            {

    [Test]
    public void WeaponsConfig_UpgradeCosts_NonNegative()

        {

    [Test]

    [Test]
    public void SchoolsConfig_PlayerBase_HP_Positive()
    {

    [Test]
    public void SchoolsConfig_Daoist_StaminaCost_IsHalf()
    {
        var config = _loader.GetConfig<SchoolConfigCollection>();
        var daoist = config.schools.First(s => s.schoolId == "Taoist");

    // ===== bosses.json 校验 =====

    [Test]
    public void BossesConfig_AllBosses_HaveThreePhases()

        foreach (var boss in config.bosses)
        {
            Assert.AreEqual(3, boss.phases.Length,

    }

    [Test]

    {

        var bullets = _loader.GetConfig<BulletConfigCollection>();

        {

            {

    public void BossesConfig_PhaseHpThresholds_Descending()

using NUnit.Framework;

/// <summary>
/// 战斗公式测试 — 对应 Story: S15
/// 验证伤害公式 5 层计算、反弹公式、边界值
/// </summary>
[TestFixture]
public class CombatFormulaTests
{
    private CombatSystem _combat;
    private SchoolConfigCollection _schools;

    [SetUp]
    public void Setup()
    {
        var loader = new ConfigLoader();
        loader.SetConfigRoot("Assets/Configs/");
        _schools = loader.GetConfig<SchoolConfigCollection>();
        _combat = new CombatSystem(loader);
    }

    // ===== 5 层伤害公式 =====

    [Test]
    public void DamageCalculation_BaseDamage_Only()
    {
        // 基础伤害，无任何系数
        var req = new DamageRequest
        {
            baseDamage = 15f,
            school = SchoolType.Confucian,  // 系数 1.0
            weaponLevel = 1,                 // 等级系数 1.0
            hasItemBonus = false,            // 器物系数 1.0
            isBossTarget = false,            // Boss系数 1.0
            shapeCoefficient = 1.0f          // 形状系数 1.0
        };
        var result = _combat.Calculate(req);
        Assert.AreEqual(15f, result.finalDamage, 0.01f);
    }

    [Test]
    public void DamageCalculation_Legalist_AttackBonus()
    {
        // 法家攻击 +10%
        var req = new DamageRequest
        {
            baseDamage = 15f,
            school = SchoolType.Legalist,    // 系数 1.1
            weaponLevel = 1,
            hasItemBonus = false,
            isBossTarget = false,
            shapeCoefficient = 1.0f
        };
        var result = _combat.Calculate(req);
        Assert.AreEqual(16.5f, result.finalDamage, 0.01f); // 15 × 1.1
    }

    [Test]
    public void DamageCalculation_Daoist_AttackPenalty()
    {
        // 道家攻击 -5%
        var req = new DamageRequest
        {
            baseDamage = 15f,
            school = SchoolType.Taoist,      // 系数 0.95
            weaponLevel = 1,
            hasItemBonus = false,
            isBossTarget = false,
            shapeCoefficient = 1.0f
        };
        var result = _combat.Calculate(req);
        Assert.AreEqual(14.25f, result.finalDamage, 0.01f); // 15 × 0.95
    }

    [Test]
    public void DamageCalculation_BossTarget_DoublesDamage()
    {
        // Boss单体 ×2
        var req = new DamageRequest
        {
            baseDamage = 15f,
            school = SchoolType.Confucian,
            weaponLevel = 1,
            hasItemBonus = false,
            isBossTarget = true,             // Boss系数 2.0
            shapeCoefficient = 1.0f
        };
        var result = _combat.Calculate(req);
        Assert.AreEqual(30f, result.finalDamage, 0.01f); // 15 × 2.0
    }

    [Test]
    public void DamageCalculation_BossTarget_DoesNotApplyToMinions()
    {
        // Boss波弟子不享受 ×2（v0.2 §E.8）
        var req = new DamageRequest
        {
            baseDamage = 15f,
            school = SchoolType.Confucian,
            weaponLevel = 1,
            hasItemBonus = false,
            isBossTarget = false,            // 不是Boss实体
            shapeCoefficient = 1.0f,
            isBossWaveMinion = true          // 但在Boss波中
        };
        var result = _combat.Calculate(req);
        Assert.AreEqual(15f, result.finalDamage, 0.01f); // 不×2
    }

    [Test]
    public void DamageCalculation_AllFiveLayers_Combined()
    {
        // 全部 5 层系数叠加
        var req = new DamageRequest
        {
            baseDamage = 15f,
            school = SchoolType.Legalist,    // 1.1
            weaponLevel = 4,                 // 假设等级系数 1.5

        var result = _combat.Calculate(req);
